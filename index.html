/**
 * Calculates POS order totals according to UKK MUK specifications.
 * 
 * @param {Array} items - List of cart items [{price, qty}]
 * @param {number} discountPercent - Tiered discount rate (0 - 100)
 * @param {number} taxRate - Tax percentage (default 0.10 for 10%)
 * @returns {Object} { subtotal, discountVal, taxVal, grandTotal }
 */
function calculatePOSOrder(items, discountPercent = 0, taxRate = 10) {
    const subtotal = items.reduce((sum, item) => sum + (item.price * item.qty), 0);
    const discountVal = subtotal * (discountPercent / 100);
    const taxableAmount = Math.max(0, subtotal - discountVal);
    const taxVal = taxableAmount * (taxRate / 100);
    const grandTotal = taxableAmount + taxVal;

    return { subtotal, discountVal, taxVal, grandTotal };
}
